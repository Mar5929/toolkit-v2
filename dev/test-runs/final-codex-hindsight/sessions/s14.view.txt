A tiny command that prints a greeting.

Run it with an optional name. With no name, it greets the world.

```text
$ node bin/greet.mjs Mike
Hello, Mike!
```

## Flags

`--times N` prints the same greeting N times, one per line. N must be a positive whole number. The name can appear before or after the flag.

```text
$ node bin/greet.mjs Mike --times 3
Hello, Mike!
Hello, Mike!
Hello, Mike!
```

Run the tests with `npm test`.

// greet: prints a friendly greeting.
// Usage: greet [name] [--times N]
export function greet(name = 'world') {
  return `Hello, ${name}!`;
}

function parseArgs(args) {
  let name;
  let times = 1;
  let seenTimes = false;

  for (let index = 0; index < args.length; index += 1) {
    const arg = args[index];
    if (arg === '--times') {
      if (seenTimes) throw new Error('--times may only be given once');
      seenTimes = true;
      const value = args[++index];
      if (!value || !/^\d+$/.test(value) || !Number.isSafeInteger(Number(value)) || Number(value) < 1) {
        throw new Error('--times requires a positive whole number');
      }
      times = Number(value);
    } else if (arg.startsWith('-') || name !== undefined) {
      throw new Error(`Unexpected argument: ${arg}`);
    } else {
      name = arg;
    }
  }

  return { name, times };
}

if (import.meta.url === `file://${process.argv[1]}`) {
  try {
    const { name, times } = parseArgs(process.argv.slice(2));
    const message = greet(name);
    for (let index = 0; index < times; index += 1) console.log(message);
  } catch (error) {
    console.error(`greet: ${error.message}`);
    process.exitCode = 1;
  }
}
