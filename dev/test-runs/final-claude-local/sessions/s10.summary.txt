#!/usr/bin/env node
// Greets the name given after the command, or the world when none is given.
// Usage: node greet.js [name...]

function greeting(args) {
  const name = args.join(' ');
  return `Hello, ${name.trim() === '' ? 'world' : name}!`;
}

if (require.main === module) {
  console.log(greeting(process.argv.slice(2)));
}

module.exports = { greeting };
