'use strict';

const USAGE = 'Usage: node slug.js "<title>"';
const MAX_ERROR = 'Error: --max needs a whole number greater than 0';

// Letters that Unicode decomposition does not split into a plain letter and an accent.
const SPECIAL_LETTERS = {
  ß: 'ss', æ: 'ae', œ: 'oe',
  ø: 'o', ł: 'l', đ: 'd', ħ: 'h', ŧ: 't', ƀ: 'b', ƶ: 'z', ǥ: 'g', ɨ: 'i', ƚ: 'l',
};
const SPECIAL_LETTER_PATTERN = new RegExp(`[${Object.keys(SPECIAL_LETTERS).join('')}]`, 'g');

function slugify(title) {
  return title
    .toLowerCase()
    .replace(SPECIAL_LETTER_PATTERN, (letter) => SPECIAL_LETTERS[letter])
    .normalize('NFKD')
    .replace(/\p{M}/gu, '')
    .replace(/['’]/g, '')
    .replace(/[^a-z0-9]+/g, '-')
